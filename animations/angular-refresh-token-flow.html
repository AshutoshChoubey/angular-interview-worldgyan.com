export const refreshTokenInterceptor: HttpInterceptorFn = (req, next) => { const authService = inject(AuthService); return next(req).pipe( catchError((error: HttpErrorResponse) => { if (error.status === 401 && !req.url.includes('/auth/refresh')) { return authService.refreshToken().pipe( switchMap((newTokens) => { const cloned = req.clone({ setHeaders: { Authorization: `Bearer ${newTokens.accessToken}` } }); return next(cloned); // Seamless automatic replay! }) ); } return throwError(() => error); }) ); };
